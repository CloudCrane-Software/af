### 4.2 各 step 的 run 体（带行号）
58:    steps:
59:      - name: 分工声明（F1 快速门在此跑；F2 全量门不走 Actions）
60:        run: |
61:          echo "F1 快速门：fmt --check / clippy -D warnings / af-dsl+af-compiler 快速测试"
62:          echo "F2 全量门：ci/run.sh 十四步——走收件箱派单+独立 runner（integrator 复核制），不进 Actions"
63:
64:      - name: checkout（PR 头树；零外部 action 依赖，不用 actions/checkout）
65:        env:
66:          AF_HEAD_REF: ${{ gitea.event.pull_request.head.ref }}
67:        run: |
68:          set -eu
69:          REF="${AF_HEAD_REF:-${GITHUB_HEAD_REF:-}}"
70:          if [ -z "$REF" ]; then
71:            echo "取不到 PR 头分支（事件上下文缺 pull_request.head.ref）" >&2
72:            exit 1
73:          fi
74:          HOST="${GITHUB_SERVER_URL#http://}"; HOST="${HOST#https://}"
75:          AUTH=""
76:          if [ -n "${AF_TOKEN:-}" ]; then
77:            AUTH="x-access-token:${AF_TOKEN}@"
78:          fi
79:          git clone --depth 1 --branch "$REF" \
80:            "http://${AUTH}${HOST}/${GITHUB_REPOSITORY}.git" .
81:          git --no-pager log --oneline -1
82:
83:      - name: 工具链供给（runner 容器无 rust——按仓内钉住版自举；已装则直接复用）
84:        run: |
85:          set -eu
86:          export PATH="$HOME/.cargo/bin:$PATH"
87:          if command -v cargo >/dev/null 2>&1; then
88:            echo "cargo 已在（复用既有供给）: $(cargo --version)"
89:          else
90:            echo "runner 容器无 rust（OpenCloudOS 9.6 极简镜像，run143/144 诊断在案）——rustup 自举"
91:            curl -sSfL https://static.rust-lang.org/rustup/dist/x86_64-unknown-linux-gnu/rustup-init \
92:              -o /tmp/rustup-init
93:            chmod +x /tmp/rustup-init
94:            /tmp/rustup-init -y --profile minimal --no-modify-path
95:          fi
96:          export PATH="$HOME/.cargo/bin:$PATH"
97:          # 按 rust-toolchain.toml 钉住的 channel（1.99.1）装工具链与 rustfmt/clippy 组件
98:          CHANNEL="$(sed -n 's/^channel *= *"\(.*\)"/\1/p' rust-toolchain.toml)"
99:          echo "钉住 channel: $CHANNEL"
100:          rustup toolchain install "$CHANNEL" --profile minimal \
101:            --component rustfmt --component clippy 2>&1 | tail -5
102:          rustup default "$CHANNEL"
103:          echo "$HOME/.cargo/bin" >> "$GITHUB_PATH"
104:          cargo --version && rustc --version && cargo fmt --version
105:
106:      - name: fmt --check（硬门；主线已归零，红即如实亮红）
107:        run: |
108:          export PATH="$HOME/.cargo/bin:$PATH"
109:          cargo fmt --check --all
110:
111:      - name: clippy -D warnings（硬门；快速门用 -D 全量拦，比 F2 的基线门更严）
112:        # fmt 红不遮蔽 clippy/test 判定——一次 run 收全三路诊断（对齐 ci/run.sh
113:        # 「步骤失败不中断、跑完全部步骤收集完整诊断」哲学）；job 结论仍任一红即红。
114:        if: ${{ always() }}
115:        run: |
116:          export PATH="$HOME/.cargo/bin:$PATH"
117:          cargo clippy --workspace --all-targets -- -D warnings
118:
119:      - name: 快速测试（af-dsl + af-compiler；封闭子集，不碰 F2 资产面）
120:        if: ${{ always() }}
121:        run: |
122:          export PATH="$HOME/.cargo/bin:$PATH"
123:          cargo test -p af-dsl -p af-compiler
